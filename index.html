# Clean up old files in ~/Downloads
import os, time

def clean_old(path, days=30):
    now = time.time()
    for f in os.listdir(path):
        fp = os.path.join(path, f)
        if os.path.getmtime(fp) < now - days * 86400:
            os.remove(fp)
            print(f"Removed: {f}")

if __name__ == "__main__":
    clean_old(os.path.expanduser("~/Downloads"))
